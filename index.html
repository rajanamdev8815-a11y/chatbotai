import streamlit as st from groq import Groq from dotenv import load_dotenv import os load_dotenv() st.set_page_config( page_title="My AI Chatbot", page_icon="🤖" ) st.title("🤖 My AI Chatbot") st.write("नमस्ते! मुझसे कुछ भी पूछिए।") api_key = os.getenv("GROQ_API_KEY") if not api_key: st.error("Groq API Key नहीं मिली।") st.stop() client = Groq(api_key=api_key) if "messages" not in st.session_state: st.session_state.messages = [] for msg in st.session_state.messages: with st.chat_message(msg["role"]): st.write(msg["content"]) prompt = st.chat_input("अपना सवाल लिखें...") if prompt: st.session_state.messages.append({ "role": "user", "content": prompt }) with st.chat_message("user"): st.write(prompt) with st.chat_message("assistant"): with st.spinner("AI जवाब दे रहा है..."): try: response = client.chat.completions.create( model="llama-3.3-70b-versatile", messages=[ { "role": "system", "content": "आप एक मददगार AI Assistant हैं। यूज़र को उसकी भाषा में जवाब दें।" }, *st.session_state.messages ] ) answer = response.choices[0].message.content st.write(answer) st.session_state.messages.append({ "role": "assistant", "content": answer }) except Exception as e: st.error(f"Error: {e}") if st.button("चैट साफ करें"): st.session_state.messages = [] st.rerun()
